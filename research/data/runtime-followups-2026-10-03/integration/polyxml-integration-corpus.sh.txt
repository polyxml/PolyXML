#!/usr/bin/env bash
set -euo pipefail
root=/home/bailey/github/PolyXML
repo=/home/bailey/github/polyxml-runtime-integration
companion=/home/bailey/github/polyxml-w3c-tests
export POLYXML_MEMCAP_BACKEND=systemd CARGO_BUILD_JOBS=1
export CARGO_TARGET_DIR="$root/target/followup-runtime-integration-verification"
mkdir -p /tmp/polyxml-followup-integration-bin
cp "$CARGO_TARGET_DIR/debug/polyxml" /tmp/polyxml-followup-integration-bin/polyxml
export PATH=/tmp/polyxml-followup-integration-bin:$PATH
test ! -e "$root/target/release/polyxml"
"$root/scripts/memcap.sh" uv --directory "$companion" run --no-sync "$root/.venv/bin/maturin" develop --manifest-path "$repo/crates/polyxml-python/Cargo.toml" > /tmp/polyxml-runtime-integration-companion-binding.log 2>&1
uv --directory "$companion" run --no-sync python /tmp/polyxml-probe-integration.py > /tmp/polyxml-runtime-integration-probe.json 2> /tmp/polyxml-runtime-integration-probe.log
"$root/scripts/memcap.sh" uv --directory "$companion" run --no-sync runner.py --suite CType --limit 50 > /tmp/polyxml-runtime-integration-CType.log 2>&1
"$root/scripts/memcap.sh" uv --directory "$companion" run --no-sync runner.py --suite AttrDecl --limit 50 > /tmp/polyxml-runtime-integration-AttrDecl.log 2>&1
"$root/scripts/memcap.sh" uv --directory "$companion" run --no-sync runner.py --suite NIST --limit 100 > /tmp/polyxml-runtime-integration-NIST.log 2>&1
