repo=Path(sys.argv[1]).resolve();label=sys.argv[2];out=Path(sys.argv[3]);out.mkdir(parents=True,exist_ok=True)

source=Path('/home/bailey/github/polyxml-mixed-nil/benchmarks/rust-runtime-investigation/mixed.rs').read_text()

harness='use polyxml::schema::ModelSchema;\nuse polyxml::schema_parser::XsdParser;\nuse polyxml::{deserialize, serialize, PolyValue};\nuse std::{hint::black_box,sync::Arc};\n'+fixture+r'''
#[inline(never)]
#[no_mangle]
fn measured_region(schema: &ModelSchema, value: &PolyValue, count: usize) {
 for _ in 0..count { black_box(serialize("Root", black_box(value), black_box(schema), None).unwrap()); }
}
fn main() {
 let kind=std::env::args().nth(1).unwrap();
 let (schema,xml)=fixture(&kind,1000);
 let value=deserialize(&xml,Arc::clone(&schema)).unwrap();
 for _ in 0..3 { black_box(serialize("Root", &value, &schema,None).unwrap()); }
 measured_region(&schema,&value,25);
}
'''
