#!/usr/bin/env bash
set -euo pipefail
cd /home/bailey/github/polyxml-mixed-index
export POLYXML_MEMCAP_BACKEND=systemd CARGO_BUILD_JOBS=1 DOTNET_ROLL_FORWARD=Major
export PATH=/tmp/polyxml-corpus-tools/node_modules/.bin:$PATH
root=/home/bailey/github/PolyXML
ln -s "$root/benchmarks/rust-xml-regression/target" benchmarks/rust-xml-regression/target
python3 benchmarks/rust-xml-regression/run_core.py --baseline ../polyxml-perf-selected --harness /tmp/polyxml-mixed-branch-concentrated.rs --output "$root/benchmarks/rust-runtime-investigation/target/evidence/mixed-branch-index-concentrated-final-sampled" --rounds 2 --samples 50 --warmup 1 --measurement 2 --filter '^branch_write/'

python3 benchmarks/rust-xml-regression/run_core.py --baseline ../polyxml-perf-selected --harness benchmarks/rust-runtime-investigation/mixed.rs --output "$root/benchmarks/rust-runtime-investigation/target/evidence/mixed-branch-index-common-sampled" --rounds 3 --samples 100 --warmup 3 --measurement 5 --filter '^mixed_write/'

python3 /tmp/polyxml-branch-allocations.py . current "$root/benchmarks/rust-runtime-investigation/target/evidence/mixed-branch-index-allocations-sampled"

ln -s "$root/.venv" .venv
scripts/memcap.sh scripts/gate.sh > /tmp/polyxml-mixed-index-gate-sampled.log 2>&1

unlink .venv
